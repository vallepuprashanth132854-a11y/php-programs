    $username = $_COOKIE["username"];
}

if(isset($_COOKIE["password"]))
{
    $password = $_COOKIE["password"];
}

if(isset($_POST["btn"]))
{
    setcookie("username", $_POST["username"], time()+3600);
    setcookie("password", $_POST["password"], time()+3600);


}

?>
